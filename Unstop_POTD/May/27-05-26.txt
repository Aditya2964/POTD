class Solution {
  public:
    bool wifiRange(string &s, int x) {
        int n = s.length();
        int covered = -1;

        for (int i = 0; i < n; i++) {
            if (s[i] == '1') {
                if (i - x > covered + 1) {
                    return false;
                }
                covered = max(covered, i + x);
            }
        }

        return covered >= n - 1;
    }
};
